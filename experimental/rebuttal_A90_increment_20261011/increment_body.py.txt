def interpretation_guard(panels):
    expected={
        ('native',10):['lower','lower','lower'],('native',9):['lower','lower','lower'],('native',6):['lower','higher','lower'],
        ('shared_calibration',10):['lower','lower','lower'],('shared_calibration',9):['lower','lower','lower'],('shared_calibration',6):['lower','higher','lower'],
        ('raw',10):['higher','lower','lower'],('raw',9):['higher','lower','lower'],('raw',6):['lower','lower','lower'],
    }
    require(len(panels)==9 and {(p['view'],p['seed_count']) for p in panels}==set(expected),'Nine S-DFA interpretation bindings')
    for p in panels:require(p['attack']=='S-DFA' and p['directions']==expected[(p['view'],p['seed_count'])],'Actual signs must support the bounded prose')
    return [p for p in panels if p['seed_count']==10]
